// Add student
export async function addStudent(reg, name, className, pass) {
    const { data, error } = await supabase
        .from('students')
        .insert([{registration_number: reg, full_name: name, class: className, password: pass, results: ''}])
    if(error) alert(error.message)
    else alert('Student added!')
}

// Get all students (Admin dashboard)
export async function getStudents() {
    const { data, error } = await supabase.from('students').select('*')
    if(error) alert(error.message)
    else return data
}

// Update results
export async function updateResults(reg, newResults) {
    const { data, error } = await supabase.from('students')
        .update({results: newResults})
        .eq('registration_number', reg)
    if(error) alert(error.message)
    else alert('Results updated!')
}

// Student Login
export async function studentLogin(reg, pass) {
    const { data, error } = await supabase
        .from('students')
        .select('*')
        .eq('registration_number', reg)
        .eq('password', pass)
        .single()
    if(error || !data) alert('Invalid registration number or password')
    else {
        localStorage.setItem('student', JSON.stringify(data))
        window.location.href = 'student-dashboard.html'
    }
}

// Get student info
export function getStudent() {
    return JSON.parse(localStorage.getItem('student'))
}
